/* CONFIGURAÇÃO MQTT */ // Mesmo broker utilizado no ESP32 const broker = "wss://broker.hivemq.com:8884/mqtt"; // Tópicos MQTT const topicoTemperatura = "senai/aula01/esp32/temperatura/Silva"; const topicoUmidade = "senai/aula01/esp32/umidade/Silva"; const topicoLed = "senai/aula01/esp32/led/Silva"; // Estados dos LEDs let led1Ligado = false; let led2Ligado = false; // Elementos da página const statusBroker = document.getElementById("statusBroker"); const textoStatus = document.getElementById("textoStatus"); const temperatura = document.getElementById("temperatura"); const umidade = document.getElementById("umidade"); const botaoLed1 = document.getElementById("botaoLed1"); const botaoLed2 = document.getElementById("botaoLed2"); // Conexão com o broker const cliente = mqtt.connect(broker, { clientId: "Dashboard_Ana_" + Math.random().toString(16).slice(2), reconnectPeriod: 2000, connectTimeout: 10000 }); // CONECTADO cliente.on("connect", () => { console.log("Conectado ao broker MQTT!"); statusBroker.className = "status-broker conectado"; textoStatus.textContent = "Conectado ao broker"; // Inscrição nos tópicos cliente.subscribe([ topicoTemperatura, topicoUmidade ], (erro) => { if (erro) { console.error("Erro ao assinar tópicos:", erro); } }); }); // RECEBIMENTO DAS MENSAGENS cliente.on("message", (topico, mensagem) => { const valor = mensagem.toString(); console.log("Tópico:", topico, "Mensagem:", valor); if (topico === topicoTemperatura) { temperatura.textContent = valor; } if (topico === topicoUmidade) { umidade.textContent = valor; } }); // CONEXÃO PERDIDA cliente.on("close", () => { statusBroker.className = "status-broker desconectado"; textoStatus.textContent = "Desconectado do broker"; }); // RECONECTANDO cliente.on("reconnect", () => { statusBroker.className = "status-broker conectando"; textoStatus.textContent = "Reconectando ao broker..."; }); // ERRO DE CONEXÃO cliente.on("error", (erro) => { console.error("Erro MQTT:", erro); }); // CONTROLE DO LED 1 function controlarLed1() { if (!cliente.connected) { alert("O Dashboard não está conectado ao broker!"); return; } const comando = led1Ligado ? "LED1_OFF" : "LED1_ON"; cliente.publish(topicoLed, comando, { qos: 1 }, (erro) => { if (erro) { console.error("Erro ao enviar comando:", erro); return; } led1Ligado = !led1Ligado; botaoLed1.textContent = led1Ligado ? "Desligar LED 1" : "Ligar LED 1"; }); } // CONTROLE DO LED 2 function controlarLed2() { if (!cliente.connected) { alert("O Dashboard não está conectado ao broker!"); return; } const comando = led2Ligado ? "LED2_OFF" : "LED2_ON"; cliente.publish(topicoLed, comando, { qos: 1 }, (erro) => { if (erro) { console.error("Erro ao enviar comando:", erro); return; } led2Ligado = !led2Ligado; botaoLed2.textContent = led2Ligado ? "Desligar LED 2" : "Ligar LED 2"; }); }
